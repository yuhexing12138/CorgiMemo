- [ ] **Step 1.2: 在 SwipeableImageStack 主函数外部（与 CardSlot 同层）添加 CardTarget + lerp helper + calcCardTarget 纯函数**

  插入位置建议：`private data class CardSlot` 之后（约 L90 附近）。完整代码：
  ```kotlin
  /**
   * 卡片在任意时刻的统一目标值（等价原型 getCardTarget 返回值）
   * 单一来源：堆叠态、展开态、过渡中间态都由 calcCardTarget 统一返回
   */
  private data class CardTarget(
      val x: Dp,
      val y: Dp,
      val rotationZ: Float,
      val scale: Float,
      val shadowElevation: Dp,
      val zIndex: Float,
  )

  /** Dp 线性插值工具 */
  private fun lerp(a: Dp, b: Dp, p: Float): Dp =
      (a.value + (b.value - a.value) * p.coerceIn(0f, 1f)).dp

  /**
   * 单一目标函数（等价原型 expand-collapse-stack.html L356 getCardTarget）
   * - expandProgress = 0 → 堆叠态
   * - expandProgress = 1 → 展开态（S2：按 displayIndex=order 顺序横排，保留翻牌进度）
   * - expandProgress ∈ (0,1) → 线性插值（400ms 补间每帧调用）
   * - displayIndex：当前 order 列表中的索引（即 ei，扇形层级；展开态 x 基准）
   */
  private fun calcCardTarget(
      displayIndex: Int,
      cardCount: Int,
      expandProgress: Float,
      cardW: Dp,
      cardGap: Dp,
      visibleDepth: Int,
      stackOffsetDp: Dp,
      fanAngleDeg: Float,
      scaleStep: Float,
  ): CardTarget {
      val ei = min(displayIndex, visibleDepth - 1)
      val denom = max(visibleDepth - 1, 1)

      val stacked = CardTarget(
          x = 0.dp,
          y = (-ei * stackOffsetDp.value).dp,
          rotationZ = fanAngleDeg * (ei.toFloat() / denom.toFloat()),
          scale = 1f - ei * scaleStep,
          shadowElevation = if (displayIndex == 0) 8.dp else 4.dp,
          zIndex = (visibleDepth - ei).toFloat()
      )

      val expanded = CardTarget(
          x = displayIndex.toFloat() * (cardW.value + cardGap.value).dp,
          y = 0.dp,
          rotationZ = 0f,
          scale = 1f,
          shadowElevation = 4.dp,
          zIndex = 1f
      )

      val p = expandProgress.coerceIn(0f, 1f)
      return CardTarget(
          x = lerp(stacked.x, expanded.x, p),
          y = lerp(stacked.y, expanded.y, p),
          rotationZ = stacked.rotationZ + (expanded.rotationZ - stacked.rotationZ) * p,
          scale = stacked.scale + (expanded.scale - stacked.scale) * p,
          shadowElevation = lerp(stacked.shadowElevation, expanded.shadowElevation, p),
          // zIndex 离散（无插值语义），阈值 0.5 切换
          zIndex = if (p < 0.5f) stacked.zIndex else expanded.zIndex
      )
  }
  ```

- [ ] **Step 1.3: 在主函数 remember 变量区（约 L280-L360，TRANSITION_SPRING/BOUNCE_SPRING 之后）添加 400ms/200ms spec + expandProgress Animatable + 派生状态**

  插入位置：`val scope = rememberCoroutineScope()` 之后、`val stageBoxWidthPx = ...` 之前。完整代码：

  ```kotlin
      // ============================================================
      // 新动画系统：单 expandProgress Animatable 驱动（P1 新增声明，P4 才接入行为）
      // ============================================================
      /** 400ms 贝塞尔补间（严格对齐原型 CSS cubic-bezier(0.22,1,0.36,1)） */
      val TRANSITION_400_SPEC = tween<Float>(
          durationMillis = 400,
          easing = CubicBezierEasing(0.22f, 1f, 0.36f, 1f)
      )
      /** 200ms 透明度补间（严格对齐原型 opacity 0.2s） */
      val OPACITY_200_SPEC = tween<Float>(durationMillis = 200)

      /**
       * 全局展开进度（核心驱动）：
       *   0f = 完全堆叠；1f = 完全展开；(0,1) = 过渡中
       * P1 仅初始化；P4 才把 setExpanded 入口接入 animateTo
       */
      val initialExpanded = remember { expandedState?.value ?: isExpanded }
      val expandProgress = remember { Animatable(initialValue = if (initialExpanded) 1f else 0f) }

      /** 过渡动画进行中（手势互斥判断） */
      val isAnimating by remember { derivedStateOf { expandProgress.isRunning } }
      /** 是否处于完全堆叠态（用于扇形拖拽手势挂载判断） */
      val isInStackedMode by remember { derivedStateOf { expandProgress.value < 0.01f && !isAnimating } }
      /** 是否处于完全展开态（用于 horizontalScroll ScrollArea 挂载判断） */
      val isInExpandedMode by remember { derivedStateOf { expandProgress.value > 0.99f && !isAnimating } }
      /** 对外兼容用 Boolean 快照（阈值 0.5，防抖回调） */
      val derivedIsExpanded by remember { derivedStateOf { expandProgress.value >= 0.5f } }
  ```

  插入位置：remember 变量区。完整代码：
  ```kotlin
      /** 上一次 onExpandStateChange 回调的 Boolean 值（防抖，避免过渡中重复回调） */
      var lastCallbackExpanded by remember { mutableStateOf(initialExpanded) }

      /**
       * 统一状态切换入口（替换所有 isExpanded = xx 的直接赋值）
       * 顺序：拖拽归零/横滚归零 → 400ms 进度补间 → 阈值跨越回调
       */
      suspend fun setExpanded(expanded: Boolean) {
          val target = if (expanded) 1f else 0f
          if (kotlin.math.abs(expandProgress.value - target) < 0.001f) return

          // --- 展开：堆叠态 dragOffset 并行归零 ---
          if (expanded) {
              coroutineScope {
                  awaitAll(
                      async { dragOffsetX.animateTo(0f, TRANSITION_SPRING) },
                      async { dragOffsetY.animateTo(0f, BOUNCE_SPRING) },
                  )
              }
          } else {
              // --- 收起：横滚归零 ---
              expandedScrollState.animateScrollTo(0)
          }

          // --- 400ms 主进度补间 ---
          expandProgress.animateTo(target, TRANSITION_400_SPEC)

          // --- 同步 isExpanded + expandedState + 回调 ---
          val effExpanded = target >= 0.5f
          isExpanded = effExpanded  // 同步原有 MutableState（保证非动画场景读取一致）
          if (expandedState?.value != effExpanded) expandedState?.value = effExpanded
          if (effExpanded != lastCallbackExpanded) {
              lastCallbackExpanded = effExpanded
              onExpandStateChange?.invoke(effExpanded)
          }
      }
  ```
